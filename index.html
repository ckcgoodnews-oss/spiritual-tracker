async function commitStateToGitHub() { const owner = document.getElementById('git-repo-owner').value.trim(); const repo = document.getElementById('git-repo-name').value.trim(); const branch = document.getElementById('git-repo-branch').value.trim() || 'main'; const token = document.getElementById('git-repo-token').value.trim(); if (!owner || !repo || !token) { alert('Please enter your GitHub Repo Owner, Repo Name, and Personal Access Token.'); return; } const filePath = 'accounts.json'; const url = `https://api.github.com/repos/${owner}/${repo}/contents/${filePath}`; // Save credentials locally for convenience localStorage.setItem('git_sync_config', JSON.stringify({ owner, repo, branch, token })); try { // 1. Get current file SHA if it exists let sha = ''; const getRes = await fetch(`${url}?ref=${branch}`, { headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github+json' } }); if (getRes.ok) { const fileData = await getRes.json(); sha = fileData.sha; } // 2. Encode JSON content to Base64 const contentPayload = JSON.stringify(state, null, 2); const base64Content = btoa(unescape(encodeURIComponent(contentPayload))); // 3. Commit file via PUT request const putRes = await fetch(url, { method: 'PUT', headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github+json', 'Content-Type': 'application/json' }, body: JSON.stringify({ message: `Admin Sync: Updated records for ${state.accounts.length} users`, content: base64Content, sha: sha || undefined, branch: branch }) }); if (putRes.ok) { alert('✅ Successfully committed state directly to GitHub repository!'); } else { const err = await putRes.json(); alert(`GitHub API Error: ${err.message}`); } } catch (e) { alert(`Failed to connect to GitHub: ${e.message}`); } } async function pullStateFromGitHub() { const owner = document.getElementById('git-repo-owner').value.trim(); const repo = document.getElementById('git-repo-name').value.trim(); const branch = document.getElementById('git-repo-branch').value.trim() || 'main'; const token = document.getElementById('git-repo-token').value.trim(); if (!owner || !repo || !token) { alert('Please enter Owner, Repo, and Token to pull data.'); return; } try { const res = await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/accounts.json?ref=${branch}`, { headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github+json' } }); if (res.ok) { const data = await res.json(); const decoded = decodeURIComponent(escape(atob(data.content))); const remoteState = JSON.parse(decoded); state = Object.assign(state, remoteState); saveToStorage(); renderUserAdminSection(); renderAdminParticipantSelector(); syncUIWithActiveRecord(); renderDays(); calculateSettlement(); alert('✅ Successfully pulled and synced latest data from GitHub repository!'); } else { alert('Could not find accounts.json on remote branch.'); } } catch (e) { alert(`Pull failed: ${e.message}`); } } // Auto-populate saved Git credentials on load function loadGitConfig() { const saved = localStorage.getItem('git_sync_config'); if (saved) { const cfg = JSON.parse(saved); if (document.getElementById('git-repo-owner')) document.getElementById('git-repo-owner').value = cfg.owner || ''; if (document.getElementById('git-repo-name')) document.getElementById('git-repo-name').value = cfg.repo || ''; if (document.getElementById('git-repo-branch')) document.getElementById('git-repo-branch').value = cfg.branch || 'main'; if (document.getElementById('git-repo-token')) document.getElementById('git-repo-token').value = cfg.token || ''; } }
